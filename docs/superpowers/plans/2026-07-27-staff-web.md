# 2단계: 직원용 웹(React) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 직원(접수직원/의사/관리자)이 사용하는 React 웹 앱을 만들고, 2단계 스펙에서 새로 필요해진 백엔드 3종(설정값 확장, 진료문구, 일정변경/재예약)을 1단계 FastAPI 백엔드에 추가한다.

**Architecture:** 백엔드는 1단계에서 만든 `acquire_as`/`require_role`/`AppError` 패턴을 그대로 재사용해 신규 서비스·라우터를 추가한다. 프론트엔드는 Vite+React+TypeScript SPA로, Supabase Auth로 로그인하고 역할에 따라 라우팅을 분기하며, React Query로 백엔드 REST API를 호출하고 일부 화면은 Supabase Realtime을 구독해 동기화한다.

**Tech Stack:** React, TypeScript, Vite, React Router, TanStack React Query, `@supabase/supabase-js`, FullCalendar(`@fullcalendar/react`), `@dnd-kit/core`(드래그앤드롭), Vitest + Testing Library + MSW(테스트)

## Global Constraints

- 백엔드 신규 코드는 1단계에서 만든 `backend/app/db/pool.py`(`acquire_as`), `backend/app/core/security.py`(`StaffContext`, `require_role`), `backend/app/core/errors.py`(`AppError`)를 그대로 재사용한다 — 새로 만들지 않는다
- 프론트엔드는 `frontend/` 디렉토리에 둔다(백엔드는 `backend/`)
- 화면 스타일(CSS 디자인)은 이 계획의 범위 밖이다 — 시맨틱 HTML과 최소한의 클래스명만 사용하고, 정교한 비주얼 디자인은 다루지 않는다
- 모든 API 오류 메시지는 한글로 노출한다(1단계 `AppError` 패턴 그대로)
- 여러 직원이 동시에 보는 화면(오늘의 현황/캘린더/대기목록)은 Supabase Realtime으로 동기화한다
- 동명이인 재확인 팝업은 접수직원의 환자 선택·예약확정 시점에만 적용하고, 의사 화면에는 적용하지 않는다
- 대기 순서 변경은 드래그앤드롭이며, 드롭 시 사유 입력 모달이 뜨고 취소하면 원래 위치로 되돌아간다

```python
async def reschedule_appointment(
    appointment_id: UUID,
    new_slot_id: UUID,
    staff: StaffContext,
    reason: str,
) -> None:
    if not reason.strip():
        raise AppError("일정변경 사유를 입력해야 합니다.", status_code=400)

    async with acquire_as(str(staff.auth_user_id)) as conn:
        row = await conn.fetchrow(
            "select slot_id, status from appointments where id = $1", appointment_id,
        )
        if row is None:
            raise AppError("예약을 찾을 수 없습니다.", status_code=404)

        new_slot = await conn.fetchrow(
            "select doctor_id from appointment_slots where id = $1", new_slot_id,
        )
        if new_slot is None:
            raise AppError("새 시간을 찾을 수 없습니다.", status_code=404)

        booked = await book_slot(new_slot_id, staff, conn=conn)
        if not booked:
            raise AppError("이미 예약된 시간입니다. 다른 시간을 선택하세요.", status_code=409)

        old_slot_id = row["slot_id"]
        if old_slot_id is not None:
            old_slot = await conn.fetchrow(
                "select doctor_id, slot_date from appointment_slots where id = $1", old_slot_id,
            )
            is_closed = await conn.fetchval(
                "select is_closed from doctor_schedule_exceptions where doctor_id = $1 and exception_date = $2",
                old_slot["doctor_id"], old_slot["slot_date"],
            )
            freed_status = "휴진" if is_closed else "빈시간"
            await conn.execute(
                "update appointment_slots set status = $1 where id = $2", freed_status, old_slot_id,
            )

        await conn.execute(
            "update appointments set slot_id = $1, doctor_id = $2, updated_at = now() where id = $3",
            new_slot_id, new_slot["doctor_id"], appointment_id,
        )
        await conn.execute(
            """
            insert into appointment_status_history (appointment_id, from_status, to_status, changed_by, reason)
            values ($1, $2, $2, $3, $4)
            """,
            appointment_id, row["status"], staff.id, reason,
        )
```

```python
from uuid import UUID

from app.core.security import StaffContext
from app.db.pool import acquire_as

ACTIVE_STATUSES = ["예약신청", "예약확정", "도착", "진료대기", "진료중"]


async def list_affected_appointments(doctor_id: UUID, staff: StaffContext) -> list[dict]:
    async with acquire_as(str(staff.auth_user_id)) as conn:
        rows = await conn.fetch(
            """
            select a.id, a.reason, p.name as patient_name, p.phone as patient_phone,
                   s.slot_date, s.start_time
            from appointments a
            join appointment_slots s on s.id = a.slot_id
            join patients p on p.id = a.for_patient_id
            join doctor_schedule_exceptions e
                on e.doctor_id = a.doctor_id and e.exception_date = s.slot_date
            where a.doctor_id = $1
              and a.status = any($2::text[])
              and (
                    e.is_closed
                    or s.start_time < e.override_start_time
                    or s.start_time >= e.override_end_time
              )
            order by s.slot_date, s.start_time
            """,
            doctor_id, ACTIVE_STATUSES,
        )
    return [dict(row) for row in rows]
```

```tsx
import { createContext, type ReactNode, useEffect, useState } from "react";

import { supabase } from "../lib/supabaseClient";

export type StaffInfo = {
  staffId: string;
  role: "receptionist" | "doctor" | "admin";
  departmentId: string | null;
};

type AuthContextValue = {
  staff: StaffInfo | null;
  loading: boolean;
};

export const AuthContext = createContext<AuthContextValue>({ staff: null, loading: true });

export function AuthProvider({
  children,
  apiBaseUrl,
}: {
  children: ReactNode;
  apiBaseUrl: string;
}) {
  const [staff, setStaff] = useState<StaffInfo | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    let cancelled = false;

    async function loadStaff() {
      const { data } = await supabase.auth.getSession();
      const token = data.session?.access_token;
      if (!token) {
        if (!cancelled) setLoading(false);
        return;
      }

      const response = await fetch(`${apiBaseUrl}/me`, {
        headers: { Authorization: `Bearer ${token}` },
      });
      if (response.ok) {
        const body = await response.json();
        if (!cancelled) {
          setStaff({ staffId: body.staff_id, role: body.role, departmentId: body.department_id });
        }
      }
      if (!cancelled) setLoading(false);
    }

    loadStaff();
    return () => {
      cancelled = true;
    };
  }, [apiBaseUrl]);

  return <AuthContext.Provider value={{ staff, loading }}>{children}</AuthContext.Provider>;
}
```

```typescript
import { supabase } from "../lib/supabaseClient";

const API_BASE_URL = import.meta.env.VITE_API_BASE_URL as string;

export class ApiError extends Error {}

export async function apiFetch<T>(path: string, options: RequestInit = {}): Promise<T> {
  const { data } = await supabase.auth.getSession();
  const token = data.session?.access_token;

  const response = await fetch(`${API_BASE_URL}${path}`, {
    ...options,
    headers: {
      "Content-Type": "application/json",
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
      ...options.headers,
    },
  });

  if (!response.ok) {
    const body = await response.json().catch(() => ({ detail: "요청 처리 중 오류가 발생했습니다." }));
    throw new ApiError(body.detail ?? "요청 처리 중 오류가 발생했습니다.");
  }

  if (response.status === 204) {
    return undefined as T;
  }
  return (await response.json()) as T;
}
```

```python
    phone: str


class RegisterPatientResponse(BaseModel):
    patient_id: UUID


@router.post("", response_model=RegisterPatientResponse)
```

```tsx
import { QueueAppointment, QueueRow } from "./QueueRow";

export function QueuePage() {
  const queryClient = useQueryClient();
  const { data: appointments } = useQuery({
    queryKey: ["queue"],
    queryFn: () => apiFetch<QueueAppointment[]>("/queue-today"),
  });

    await reorderQueue(appointmentId, { new_position: newPosition, reason });
    setPending(null);
    queryClient.invalidateQueries({ queryKey: ["queue"] });

    const target = appointments?.find((a) => a.id === appointmentId);

    queryClient.invalidateQueries({ queryKey: ["queue"] });

      <div role="list">
        {appointments.map((appointment, index) => (

            key={appointment.id}
            appointment={appointment}
            currentPosition={index + 1}
            pendingNewPosition={pending?.id === appointment.id ? pending.newPosition : null}
            onReorderConfirmed={handleReorderConfirmed}
            onReorderCancelled={() => setPending(null)}
```

```python
async def list_active_doctors(staff: StaffContext) -> list[dict]:
```

```tsx
import { apiFetch } from "../../api/httpClient";
import { findPatient, registerPatient } from "../../api/patients";

type Props = {

  onCancel: () => void;
};

type FoundPatient = { id: string; name: string; birth_date: string };

export function WalkInRegistrationModal({ onRegistered, onCancel }: Props) {
  const { data: doctors } = useQuery({ queryKey: ["doctors"], queryFn: listDoctors });

  const [phone, setPhone] = useState("");
  const [birthDate, setBirthDate] = useState("");
  const [searched, setSearched] = useState(false);
  const [foundPatient, setFoundPatient] = useState<FoundPatient | null>(null);
  const [confirmedPatientId, setConfirmedPatientId] = useState<string | null>(null);
  const [newName, setNewName] = useState("");
  const [newGender, setNewGender] = useState("");
  const [doctorId, setDoctorId] = useState("");
  const [showFinalConfirm, setShowFinalConfirm] = useState(false);

  async function handleSearch() {
    const result = await findPatient(phone, birthDate);
    setSearched(true);
    if (!result.patient_id) {
      setFoundPatient(null);
      return;
    }
    const detail = await apiFetch<{ id: string; name: string; birth_date: string }>(
      `/patients/${result.patient_id}`,
    );

    const result = await registerPatient({ name: newName, birth_date: birthDate, gender: newGender, phone });
    setConfirmedPatientId(result.patient_id);
  }

  async function handleAddToQueue() {
    if (!confirmedPatientId || !doctorId) return;
    const doctor = doctors?.find((d) => d.id === doctorId);
    if (!doctor) return;
    const result = await createAppointment({
      account_patient_id: confirmedPatientId,
      for_patient_id: confirmedPatientId,
      department_id: doctor.department_id,
      doctor_id: doctorId,
      reason: "당일 방문",
      source: "staff",
      initial_status: "진료대기",
    });
    setShowFinalConfirm(false);
    onRegistered(result.appointment_id);

    <div role="dialog" aria-label="당일 방문 환자 등록">
      <label htmlFor="walkin-phone-input">전화번호</label>

      {foundPatient && !confirmedPatientId && (
        <ConfirmIdentityDialog
          patientName={foundPatient.name}
          patientBirthDate={foundPatient.birth_date}
          actionLabel="환자로 선택합니다"
          onConfirm={() => setConfirmedPatientId(foundPatient.id)}
          onCancel={() => setFoundPatient(null)}
        />
      )}

      {searched && !foundPatient && !confirmedPatientId && (

          <p>조회된 환자가 없습니다. 신규 환자로 등록합니다.</p>

        </div>

          <label htmlFor="walkin-doctor-select">담당 의사</label>

              </option>
            ))}
          </select>
          <button type="button" onClick={onCancel}>
            취소
          </button>
          <button type="button" disabled={!doctorId} onClick={() => setShowFinalConfirm(true)}>
            대기 목록에 추가
          </button>
        </div>
      )}

      {/* 2차 동명이인 확인: 실제로 대기열에 등록되는 시점 (설계서 섹션 6 — "검색 선택 시점 1회 + 확정 시점 1회") */}
      {showFinalConfirm && confirmedPatientId && (
        <ConfirmIdentityDialog
          patientName={foundPatient?.name ?? newName}
          patientBirthDate={foundPatient?.birth_date ?? birthDate}
          actionLabel="이 환자를 대기 목록에 추가합니다"
          onConfirm={handleAddToQueue}
          onCancel={() => setShowFinalConfirm(false)}
        />
      )}
    </div>
  );
```

```tsx
            <QueueRow
              key={appointment.id}
              appointment={appointment}
              currentPosition={index + 1}
              pendingNewPosition={pending?.id === appointment.id ? pending.newPosition : null}
              onReorderConfirmed={handleReorderConfirmed}
              onReorderCancelled={() => setPending(null)}
              onUrgentToggled={handleUrgentToggled}
```

```tsx
import { useQuery } from "@tanstack/react-query";
import { useState } from "react";

import { listAffectedAppointments } from "../../api/appointments";

type Props = {
  doctorId: string;
  onConfirm: (selectedIds: string[]) => void;
  onCancel: () => void;
};

export function AffectedAppointmentsDialog({ doctorId, onConfirm, onCancel }: Props) {
  const { data } = useQuery({
    queryKey: ["affected-appointments", doctorId],
    queryFn: () => listAffectedAppointments(doctorId),
  });
  const [selected, setSelected] = useState<Set<string> | null>(null);

  if (!data) return <p>확인 중입니다...</p>;

  const selectedIds = selected ?? new Set(data.map((a) => a.id));
  if (selected === null && data.length > 0) setSelected(selectedIds);

  if (data.length === 0) {
    return (
      <div role="dialog">
        <p>영향받는 예약이 없습니다.</p>
        <button type="button" onClick={() => onConfirm([])}>
          저장
        </button>
      </div>
    );
  }

  function toggle(id: string) {
    const next = new Set(selectedIds);
    if (next.has(id)) next.delete(id);
    else next.add(id);
    setSelected(next);
  }

  return (
    <div role="dialog">
      <p>영향받는 예약 {data.length}건</p>
      {data.map((appointment) => (
        <label key={appointment.id}>
          <input
            type="checkbox"
            checked={selectedIds.has(appointment.id)}
            onChange={() => toggle(appointment.id)}
          />
          {appointment.patient_name} ({appointment.patient_phone}) — {appointment.slot_date} {appointment.start_time}
        </label>
      ))}
      <button type="button" onClick={onCancel}>
        취소
      </button>
      <button type="button" onClick={() => onConfirm(Array.from(selectedIds))}>
        저장
      </button>
    </div>
  );
}
```

```typescript
import { useEffect } from "react";

import { supabase } from "./supabaseClient";

export function useRealtimeSubscription(table: string, onChange: () => void) {
  useEffect(() => {
    const channel = supabase
      .channel(`realtime-${table}`)
      .on("postgres_changes", { event: "*", schema: "public", table }, onChange)
      .subscribe();

    return () => {
      supabase.removeChannel(channel);
    };
  }, [table, onChange]);
}
```

```tsx
type Props = {
  slotId: string;
  doctorId: string;
  departmentId: string;
  onBooked: (appointmentId: string) => void;
  onCancel: () => void;
};

type FoundPatient = { id: string; name: string; birth_date: string };

export function PhoneBookingModal({ slotId, doctorId, departmentId, onBooked, onCancel }: Props) {
  const [phone, setPhone] = useState("");
  const [birthDate, setBirthDate] = useState("");
  const [foundPatient, setFoundPatient] = useState<FoundPatient | null>(null);
  const [confirmedPatientId, setConfirmedPatientId] = useState<string | null>(null);
  const [reason, setReason] = useState("");
  const [showFinalConfirm, setShowFinalConfirm] = useState(false);

  async function handleSearch() {
    const result = await findPatient(phone, birthDate);
    if (!result.patient_id) {
      setFoundPatient(null);
      return;
    }
    const detail = await apiFetch<{ id: string; name: string; birth_date: string }>(
      `/patients/${result.patient_id}`,
    );
    setFoundPatient({ id: detail.id, name: detail.name, birth_date: detail.birth_date });
  }

  async function handleSave() {
    if (!confirmedPatientId) return;
    const result = await createAppointment({
      account_patient_id: confirmedPatientId,
      for_patient_id: confirmedPatientId,
      department_id: departmentId,
      doctor_id: doctorId,
      reason,
      source: "staff",
      initial_status: "예약확정",
      slot_id: slotId,
    });
    setShowFinalConfirm(false);
    onBooked(result.appointment_id);
  }

        <ConfirmIdentityDialog
          patientName={foundPatient.name}
          patientBirthDate={foundPatient.birth_date}
          actionLabel="환자로 선택합니다"
          onConfirm={() => setConfirmedPatientId(foundPatient.id)}
          onCancel={() => setFoundPatient(null)}
        />
      )}

      {confirmedPatientId && !showFinalConfirm && (
        <div>
          <label htmlFor="reason-input">방문 이유</label>

          <button type="button" onClick={onCancel}>

          </button>
          <button type="button" onClick={() => setShowFinalConfirm(true)}>
            예약 저장
          </button>
        </div>

          onCancel={() => setShowFinalConfirm(false)}
        />
      )}
```

```tsx
  if (!slots) return <p>불러오는 중입니다...</p>;

  const events = slots.map((slot) => ({
    id: slot.id,
    title: slot.status === "예약됨" ? (slot.patient_name ?? "예약됨") : slot.status,
    date: `${slot.slot_date}T${slot.start_time}`,
    classNames: [
      slot.status === "빈시간" ? "slot-empty" : slot.status === "휴진" ? "slot-closed" : "slot-booked",
      slot.has_schedule_conflict ? "slot-conflict" : "",
    ],
  }));

  return (
    <div>
      <FullCalendar
        plugins={[dayGridPlugin, timeGridPlugin, interactionPlugin]}
        initialView="timeGridWeek"
        events={events}
        dateClick={(arg: DateClickArg) => {
          const slot = slots.find((s) => s.id === arg.event?.id);
          if (slot && slot.status === "빈시간") setBookingSlot(slot);
        }}
      />
      {bookingSlot && (
        <PhoneBookingModal
          slotId={bookingSlot.id}
          doctorId={bookingSlot.doctor_id}
          departmentId={bookingSlot.department_id}
          onCancel={() => setBookingSlot(null)}
          onBooked={() => {
            setBookingSlot(null);
            queryClient.invalidateQueries({ queryKey: ["calendar-slots"] });
          }}
        />
      )}
    </div>
  );
}
```

```python
            raise AppError("취소 요청을 찾을 수 없습니다. 이미 처리되었을 수 있습니다.", status_code=404)

        await conn.execute(
            "update appointments set status = '환자취소', cancellation_requested_at = null, updated_at = now() "
            "where id = $1",
            appointment_id,
        )
        if row["slot_id"] is not None:
            await release_slot(row["slot_id"], staff, conn=conn)
        await conn.execute(
            """
            insert into appointment_status_history (appointment_id, from_status, to_status, changed_by, reason)
            values ($1, $2, '환자취소', $3, '취소 요청 승인')
            """,
            appointment_id, row["status"], staff.id,
        )


async def reject_cancellation_request(appointment_id: UUID, staff: StaffContext) -> None:
    async with acquire_as(str(staff.auth_user_id)) as conn:
        row = await conn.fetchrow(
            "select status, for_patient_id, cancellation_requested_at from appointments where id = $1",
            appointment_id,
        )
        if row is None or row["cancellation_requested_at"] is None:
            raise AppError("취소 요청을 찾을 수 없습니다. 이미 처리되었을 수 있습니다.", status_code=404)

        await conn.execute(
            "update appointments set cancellation_requested_at = null, updated_at = now() where id = $1",
            appointment_id,
        )
        await conn.execute(
            """
            insert into appointment_status_history (appointment_id, from_status, to_status, changed_by, reason)
            values ($1, $2, $2, $3, '취소 요청 반려')
            """,
            appointment_id, row["status"], staff.id,
        )
        patient_id = row["for_patient_id"]

    try:
        await notification_service.notify_patient(patient_id, "cancellation_rejected")
    except Exception:
        pass
```
